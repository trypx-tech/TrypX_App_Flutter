### 1. `lib/feature/onboarding/state/onboarding_draft.dart`
An immutable draft holding in-progress answers. Provide:
```dart
/// Mutable-via-copyWith draft of an in-progress onboarding submission.
library;

import 'package:trypx/core/models/enums.dart';
import 'package:trypx/core/models/submission.dart';

class OnboardingDraft {
  const OnboardingDraft({
    this.persona,
    this.displayName = '',
    this.social,
    this.places = const <PlaceClaim>[],
    this.languages = const <LanguageClaim>[],
    this.interviewStartsAt,
  });

  final Persona? persona;
  final String displayName;
  final SocialLink? social;
  final List<PlaceClaim> places;
  final List<LanguageClaim> languages;
  final DateTime? interviewStartsAt;

  OnboardingDraft copyWith({
    Persona? persona,
    String? displayName,
    SocialLink? social,
    bool clearSocial = false,
    List<PlaceClaim>? places,
    List<LanguageClaim>? languages,
    DateTime? interviewStartsAt,
  }) {
    return OnboardingDraft(
      persona: persona ?? this.persona,
      displayName: displayName ?? this.displayName,
      social: clearSocial ? null : (social ?? this.social),
      places: places ?? this.places,
      languages: languages ?? this.languages,
      interviewStartsAt: interviewStartsAt ?? this.interviewStartsAt,
    );
  }

  bool get hasDeepPlace =>
      places.any((p) => p.depth == PlaceDepth.livedThere || p.depth == PlaceDepth.frequent);

  bool get hasLanguage => languages.isNotEmpty;
}
```
